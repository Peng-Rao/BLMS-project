%=======================================================================
\section{Log--log Regression}
%=======================================================================

\begin{frame}{Model and reference prior}
	\textbf{Goal.} Verify the physical power law $\mathrm{resistance}\propto\Fn^{\,\gamma_F}$
	with $\gamma_F\approx4$. Taking logs makes the exponent a regression slope:
	\begin{equation*}
		\log(\mathrm{resistance}) = \beta_0 + \bF\,\log(\Fn)
		+ \sum_{k=1}^{5}\beta_k\,\mathrm{hull}_k + \varepsilon,
		\qquad \varepsilon\sim\mathcal{N}(0,\sigma^2).
	\end{equation*}
	Baseline non-informative \textbf{Jeffreys reference prior}
	\[
		\pi(\bm\beta\mid\sigma^2)\propto 1,\qquad \pi(\sigma^2)\propto\sigma^{-2}.
	\]
	\begin{block}{Closed-form posterior}
		$\bm\beta\mid\sigma^2,\bm y\sim\mathcal{N}\!\big(\hat{\bm\beta},\,\sigma^2(\bm X^{\!\top}\bm X)^{-1}\big)$;
		marginalising $\sigma^2$ gives a multivariate Student-$t$ with $n-p$ d.o.f.
		The $95\%$ credible intervals coincide with the OLS confidence intervals, no
		simulation required.
	\end{block}
\end{frame}

\begin{frame}{Result: the Froude exponent}
	\begin{columns}[c]
		\begin{column}{0.5\textwidth}
			\centering
			\includegraphics[width=\textwidth]{figures/loglog_scatter.pdf}
		\end{column}
		\begin{column}{0.5\textwidth}
			\centering
			\includegraphics[width=\textwidth]{figures/posterior_betaF.pdf}
		\end{column}
	\end{columns}
	\vspace{0.4em}
	\begin{itemize}
		\item Excellent fit: $R^2 = 0.97$, residual SD $s = 0.32$ on the log scale.
		\item $\bF \approx 4.69$, \ $95\%$ CI $=[4.60,\,4.78]$, \ $\Pr(\bF>4\mid\bm y)\approx 1$.
		\item Power law confirmed, the exponent is \emph{slightly steeper} than the textbook $\Fn^{4}$.
	\end{itemize}
\end{frame}

\begin{frame}{Hull covariates \& residual diagnostics}
	\begin{columns}[T]
		\begin{column}{0.46\textwidth}
			\centering
			\includegraphics[width=\textwidth]{figures/coef_forest.pdf}
		\end{column}
		\begin{column}{0.54\textwidth}
			\textbf{Hull form factors.} Once $\Fn$ is in the model, every $95\%$ interval
			contains zero, the factors vary only across the $22$ hulls and are strongly
			collinear (resolved by BMA in Part~3).
			\vspace{0.7em}

			\textbf{Residual diagnostics expose two defects:}
			\begin{itemize}
				\item $\cup$-shaped trend vs.\ fitted values, a single constant exponent is too rigid;
				\item spread shrinks as the fitted value grows, textbook \emph{heteroscedasticity}.
			\end{itemize}
			\vspace{0.4em}
			Both motivate the nonlinear correction in Part~4.
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Prior sensitivity: does the prior drive $\bF>4$?}
	Encode the physics in \textbf{independent informative priors} and ask how hard they
	must insist on $4$:
	\[
		\bF\sim\mathcal{N}(4,\tau^2),\quad \beta_{j\neq F}\sim\mathcal{N}(0,100^2),\quad
		\sigma^2\sim\text{Inv-Gamma}(10^{-3},10^{-3}),
	\]
	sampled with a \textbf{Gibbs sampler} ($10^4$ iters, $2\times10^3$ burn-in), sweeping $\tau$.
	\begin{columns}[c]
		\begin{column}{0.5\textwidth}
			\centering
			\includegraphics[width=\textwidth]{figures/posterior_betaF_prior.pdf}
		\end{column}
		\begin{column}{0.5\textwidth}
			\footnotesize
			\begin{tabular}{l r c}
				\toprule
				\textbf{Prior on $\bF$} & \textbf{Mean} & $\Pr(\bF>4)$ \\
				\midrule
				$\mathcal{N}(4,0.05^2)$ & $4.33$        & $1.00$       \\
				$\mathcal{N}(4,0.10^2)$ & $4.56$        & $1.00$       \\
				$\mathcal{N}(4,0.50^2)$ & $4.68$        & $1.00$       \\
				reference prior         & $4.69$        & $1.00$       \\
				\bottomrule
			\end{tabular}
			\vspace{0.6em}

			\normalsize Only a near-dogmatic $\tau=0.05$ drags the mean to $4.33$.
			For any diffuse prior ($\tau\ge0.1$) the data dominate:
			$\bF>4$ is \textbf{robust to the prior}.
		\end{column}
	\end{columns}
\end{frame}

\section{BAS / BMA Model Selection}
%=======================================================================

\begin{frame}{Model space: which hull factors matter after Froude?}
	\textbf{Conditional question.} Once the dominant physical predictor is fixed,
	which hull-geometry variables still carry signal?
	\[
		\log(\mathrm{resistance}) =
		\beta_0 + \beta_F \log(\Fn)
		+ \sum_{k \in \mathcal{M}} \beta_k\,\mathrm{hull}_k
		+ \varepsilon.
	\]
	\begin{itemize}
		\item $\log(\Fn)$ is included in every candidate model.
		\item Optional hull covariates: \texttt{LCB}, \texttt{prismatic},
		      \texttt{L\_disp}, \texttt{beam\_draught}, \texttt{L\_beam}.
		\item Five optional variables imply $2^5 = 32$ submodels.
		\item BAS/BMA averages over models instead of relying on one saturated regression.
	\end{itemize}
	\begin{block}{Question}
		Among the hull factors, which ones still matter conditional on Froude?
	\end{block}
\end{frame}

\begin{frame}{Baseline result: beam-draught has the largest PIP}
	\small
	Posterior inclusion probability (PIP) is the posterior probability that a
	covariate belongs in the regression model after averaging over all $32$ submodels.
	\vspace{0.4em}
	\begin{center}
		\begin{tabular}{lcc}
			\toprule
			\textbf{Covariate} & \textbf{PIP} & \textbf{Evidence} \\
			\midrule
			\texttt{beam\_draught} & $0.565$ & moderate \\
			\texttt{prismatic} & $0.265$ & weak \\
			\texttt{LCB} & $0.210$ & weak \\
			\texttt{L\_disp} & $0.206$ & weak \\
			\texttt{L\_beam} & $0.192$ & weak \\
			\bottomrule
		\end{tabular}
	\end{center}
	\begin{itemize}
		\item Highest posterior model: $\log(\Fn) + \texttt{beam\_draught}$.
		\item Its posterior probability is only $0.234$, so model uncertainty remains substantial.
		\item \texttt{beam\_draught} carries the strongest residual signal, but evidence is not decisive.
	\end{itemize}
	\textbf{Takeaway.} Froude is robust; hull effects are weaker, with beam-draught as the best secondary candidate.
\end{frame}

\begin{frame}{Sensitivity: is the BAS/BMA ranking robust to the prior?}
	\small
	The baseline analysis was checked under alternative BAS/BMA prior specifications,
	with the same $32$-model space and $\log(\Fn)$ forced into every model.
	\vspace{0.4em}
	\begin{center}
		\scriptsize
		\resizebox{\textwidth}{!}{%
		\begin{tabular}{lccccc}
			\toprule
			\textbf{Prior} & \textbf{LCB} & \textbf{prismatic} & \textbf{L\_disp} & \textbf{beam\_draught} & \textbf{L\_beam} \\
			\midrule
			BIC / reference & $0.210$ & $0.265$ & $0.206$ & $0.565$ & $0.192$ \\
			$g$-prior ($g=n$) & $0.186$ & $0.217$ & $0.182$ & $0.499$ & $0.166$ \\
			JZS & $0.060$ & $0.057$ & $0.051$ & $0.243$ & $0.042$ \\
			\bottomrule
		\end{tabular}%
		}
	\end{center}
	\begin{itemize}
		\item \texttt{beam\_draught} is top-ranked under all three specifications.
		\item Its PIP falls from $0.565$ to $0.243$, so selection is prior-sensitive.
		\item No hull covariate receives consistently strong support.
	\end{itemize}
	\begin{block}{Robust conclusion}
		Not ``beam-draught is definitely selected'', but ``beam-draught is the most
		defensible secondary hull covariate.''
	\end{block}
\end{frame}

\begin{frame}{From BAS/BMA to the nonlinear model}
	\[
		\begin{aligned}
			\text{Log--log regression}
			&\Longrightarrow \text{BAS/BMA over hull factors}\\
			&\Longrightarrow \text{Nonlinear models } M_0 \text{ vs. } M_1.
		\end{aligned}
	\]
	\begin{itemize}
		\item BAS/BMA proposes \texttt{beam\_draught} as the best-supported secondary hull covariate.
		\item Because evidence is moderate, we do not assume it is certainly important.
		\item Next step:
		\[
			M_0: \text{Froude only}, \qquad
			M_1: \text{Froude} + \text{\texttt{beam\_draught}}.
		\]
		\item The nonlinear heteroscedastic model checks whether this candidate improves
		      prediction on the original resistance scale.
	\end{itemize}
	\begin{block}{Takeaway}
		BAS/BMA is used as a screening step: it narrows the hull variables to the most
		defensible one, but leaves the final assessment to the nonlinear model comparison.
	\end{block}
\end{frame}

\section{Posterior Predictive Curves}
%=======================================================================

\begin{frame}{Why move back to the original resistance scale?}
	\small
	The log--log model is fitted on the transformed response,
	\[
		\log(Y_i)=x_i^{\top}\beta+\varepsilon_i,
		\qquad \varepsilon_i\sim\mathcal{N}(0,\sigma^2).
	\]
	But the measured physical quantity is resistance itself, $Y_i$, not $\log(Y_i)$.
	For prediction we therefore simulate on the log scale and transform back:
	\[
		\log(Y_i^{\mathrm{rep}})\sim\mathcal{N}(x_i^{\top}\beta,\sigma^2),
		\qquad
		Y_i^{\mathrm{rep}}=\exp\{\log(Y_i^{\mathrm{rep}})\}.
	\]
	\begin{block}{Speaking point}
		Exponentiating is not cosmetic: it returns predictions to the resistance scale,
		keeps them positive, and makes the log--log predictive distribution lognormal.
	\end{block}
\end{frame}

\begin{frame}{What is $Y^{\mathrm{rep}}$ and why is $\mu_i$ exponential?}
	\small
	Posterior predictive checking generates replicated observations from the fitted model:
	\[
		Y_i^{\mathrm{rep}}\mid\theta^{(s)} \sim p(Y_i\mid\theta^{(s)}),
	\]
	so each band includes both posterior parameter uncertainty and fresh observation noise.
	\vspace{0.4em}

	For the heteroscedastic nonlinear model we predict directly on the resistance scale:
	\[
		Y_i^{\mathrm{rep}}\sim\mathrm{Gamma}(\alpha_i,r_i),
		\qquad
		\mu_i=\exp\{b_0+\gamma_F\widetilde{\log \Fn}_i+
		\beta_B\widetilde{bd}_i\}.
	\]
	\[
		\sigma_i=\sigma_0\mu_i^{\delta},
		\qquad
		\alpha_i=(\mu_i/\sigma_i)^2,
		\quad r_i=\mu_i/\sigma_i^2.
	\]
	\begin{itemize}
		\item The exponential form is the power law $k\Fn^{\gamma_F}\exp(\beta_B bd)$ written as a positive mean.
		\item $\delta=0$ means constant variance; $\delta>0$ lets uncertainty grow with resistance.
	\end{itemize}
\end{frame}

\begin{frame}{Which models are compared?}
	\small
	\textbf{Variable selection came from BAS/BMA.}
	\begin{itemize}
		\item Froude is forced into every model because it is the dominant physical predictor.
		\item Among the five hull covariates, \texttt{beam\_draught} has the largest PIP ($\approx0.565$).
		\item Evidence is only moderate, so the nonlinear stage treats it as a candidate, not a guaranteed effect.
	\end{itemize}
	\vspace{0.3em}
	\begin{center}
		\begin{tabular}{ll}
			\toprule
			\textbf{Model} & \textbf{Purpose in predictive comparison} \\
			\midrule
			Log--log & Baseline power law, back-transformed to $Y$ \\
			Gamma M1, $\delta=0$ & Same mean, constant variance \\
			Gamma M1, $\delta$ learned & Same mean, heteroscedastic variance \\
			\bottomrule
		\end{tabular}
	\end{center}
	\begin{block}{Main question}
		Does learning heteroscedasticity improve calibration without losing the Froude power-law interpretation?
	\end{block}
\end{frame}

\begin{frame}{Posterior predictive curves and band widths}
	\begin{columns}[T]
		\begin{column}{0.50\textwidth}
			\centering
			\includegraphics[width=\textwidth]{figures/posterior_predictive_curves.pdf}
		\end{column}
		\begin{column}{0.50\textwidth}
			\centering
			\includegraphics[width=\textwidth]{figures/posterior_predictive_widths.pdf}
		\end{column}
	\end{columns}
	\vspace{0.35em}
	\scriptsize
	\begin{itemize}
		\item Left: observed resistances are black points; bands are $95\%$ posterior predictive intervals on the original scale.
		\item Log--log is positive after back-transformation and imposes multiplicative uncertainty.
		\item Gamma learned variance also stays positive, but estimates $\sigma_i=\sigma_0\mu_i^\delta$ directly on the resistance scale.
		\item Right: constant variance is too narrow; learned variance widens intervals where resistance is larger.
	\end{itemize}
\end{frame}

\begin{frame}{Calibration and train/test check}
	\small
	\begin{columns}[T]
		\begin{column}{0.48\textwidth}
			\textbf{Full-data posterior predictive calibration}
			\vspace{0.3em}
			\scriptsize
			\begin{tabular}{lcc}
				\toprule
				\textbf{Model} & \textbf{Cov.} & \textbf{Width} \\
				\midrule
				Log--log & $0.968$ & $12.241$ \\
				Gamma const. var. & $0.679$ & $4.212$ \\
				Gamma learned var. & $0.932$ & $5.403$ \\
				\bottomrule
			\end{tabular}
			\vspace{0.7em}

			\normalsize Constant variance under-covers badly; learned variance restores near-nominal coverage with narrower bands than log--log.
		\end{column}
		\begin{column}{0.52\textwidth}
			\textbf{80/20 train/test posterior predictive check}
			\vspace{0.3em}
			\scriptsize
			\begin{tabular}{lccc}
				\toprule
				\textbf{Model} & \textbf{Mean LPD} & \textbf{Cov.} & \textbf{Width} \\
				\midrule
				Log--log & $-1.540$ & $0.984$ & $14.662$ \\
				Gamma const. & $-3.773$ & $0.710$ & $4.420$ \\
				Gamma learned & $-1.992$ & $0.935$ & $6.055$ \\
				\bottomrule
			\end{tabular}
			\vspace{0.7em}

			\normalsize The split is a sanity check: log--log wins LPD but is widest; learned Gamma gives calibrated intervals on the physical scale.
		\end{column}
	\end{columns}
	\vspace{0.4em}
	\begin{block}{Takeaway}
		The robust story is the Froude power law plus heteroscedastic uncertainty; the hull effect remains weak.
	\end{block}
\end{frame}
