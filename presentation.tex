% ---- frame title: full-bleed colour bar with a thin accent rule -----
% Side margin from the text block to the paper edge (symmetric by default).
% \rlap gives the bar zero measured width, so the full bleed never triggers
% an overfull-hbox warning.
\newdimen\fullbleedoffset
\setbeamertemplate{frametitle}{%
  \fullbleedoffset=\dimexpr(\paperwidth-\textwidth)/2\relax
  \nointerlineskip
  \hspace*{-\fullbleedoffset}%
  \rlap{%
    \begin{beamercolorbox}[wd=\paperwidth,ht=2.6ex,dp=1.2ex,leftskip=\fullbleedoffset]{frametitle}%
      \usebeamerfont{frametitle}\insertframetitle%
    \end{beamercolorbox}}%
  \par\nointerlineskip
  \hspace*{-\fullbleedoffset}\rlap{\color{PoliCyan}\rule{\paperwidth}{1.2pt}}%
}
